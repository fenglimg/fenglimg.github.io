{%- comment -%} 计算第 i 张卡片（共 n 张）在 4 列网格中的尺寸，结果写入变量 bento_size。 规则：前 4 张拼成两行（大卡 + 两张小卡 + 一张中卡），之后每 3 张一行（中卡 + 两张小卡）， 最后一行不满时把剩下的卡片拉宽补齐，保证网格不留空。 用法：{% include bento/bento-size.html i=forloop.index0 n=total %} {%- endcomment -%} {%- assign _i = include.i -%} {%- assign _n = include.n -%} {%- assign _rest = _n | minus: 4 | modulo: 3 -%} {%- assign _last = _n | minus: _rest -%} {%- if _n < 4 -%} {%- if _i == 0 -%}{%- assign bento_size = "lg" -%}{%- else -%}{%- assign bento_size = "wide" -%}{%- endif -%} {%- elsif _i == 0 -%}{%- assign bento_size = "lg" -%} {%- elsif _i < 3 -%}{%- assign bento_size = "sm" -%} {%- elsif _i == 3 -%}{%- assign bento_size = "wide" -%} {%- elsif _rest == 1 and _i >= _last -%}{%- assign bento_size = "full" -%} {%- elsif _rest == 2 and _i >= _last -%}{%- assign bento_size = "wide" -%} {%- else -%} {%- assign _j = _i | minus: 4 | modulo: 3 -%} {%- if _j == 0 -%}{%- assign bento_size = "wide" -%}{%- else -%}{%- assign bento_size = "sm" -%}{%- endif -%} {%- endif -%}
